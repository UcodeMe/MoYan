+ 保留 img-ie8 文件夹。

## 二、准备一个 config.js 文件
新建一个 config.js 文件,放在 “asset/js/” 项目资源中，把下面的代码复制粘贴到，你的空文件中。
```js
(function($, window, doc){

	var CSS_PATH = "../../dist/main/css/"; // 定义框架 CSS 资源相对路径
	var SCRIPT_PATH = "../../dist/main/js/"; // 定义框架 JS 资源相对路径

	// 组件相关路径定义
	$.extend({
		UI_POPUP_JS: SCRIPT_PATH + "ui/component/base/popup.js",   // 不需要修改
		UI_DIALOG_JS: SCRIPT_PATH + "ui/component/base/dialog.js", // 不需要修改
		UI_FRAME_JS: SCRIPT_PATH + "ui/component/base/frame.js", // 不需要修改
		UI_TABSET_JS: SCRIPT_PATH + "ui/component/tabset/tabset.js", // 不需要修改

		FILE_UPLOAD_URL: "attach?action=upload", // 定义附件文件上传服务路径
		FILE_DELETE_URL: "attach?action=delete", // 定义附件文件删除服务路径
		FILE_DOWNLOAD_URL: "attach?action=download", //定义附件文件下载服务路径
		FILE_QUERY_URL: "attach?action=query", // 定义附件文件查询服务路径

		IMPEXP_URL: "impexp" //定义导入导出服务路径
	});

	// JWC 资源路径定义
	$.extend($.JWC.resource, {
		RENDER_CSS: false,
		CSS_PATH: CSS_PATH,
		SCRIPT_PATH: SCRIPT_PATH
	});

})(window.Wade, window, document);
```

## 三、JWC 实验
复制以下代码，到你的 HTML
```html
<!DOCTYPE HTML>
<html size="s">
<head>
	<meta http-equiv="X-UA-Compatible" content="IE=edge,chrome=1" />
	<meta name="viewport" content="width=device-width, initial-scale=1, minimum-scale=1, maximum-scale=1" />
	<meta http-equiv="Content-Type" content="text/html; charset=UTF-8" />
	<link href="dist/main/css/base.css" rel="stylesheet" type="text/css" />
	<script type="text/javascript" src="dist/main/js/jq/jquery-1.9.1.js"></script>
	<script type="text/javascript" src="dist/main/js/jq/adapter-1.9.1.js"></script>
	<script type="text/javascript" src="dist/main/js/jcl.js"></script>
	<script type="text/javascript" src="dist/main/js/i18n/code.zh_CN.js"></script>
	<script type="text/javascript" src="dist/main/js/jcl-plugins.js"></script>
	<script type="text/javascript" src="dist/main/js/jcl-ui.js"></script>
	<script type="text/javascript" src="dist/main/js/jcl-jwc.js"></script>
	<script type="text/javascript" src="../asset/js/config.js"></script>
	<title>test</title>
</head>
<body jwcid="@Body" id="body">
	<span jwcid="@Select"
		name="mySelect"
		source="ognl:ds">
	</span>
	<script type="text/javascript">
		// 模拟数据
		window.ds = [
			{
				TEXT: "选项（一）",
				VALUE: "value1"
			},
			{
				TEXT: "选项（二）",
				VALUE: "value2"
			},
			{
				TEXT: "选项（三）",
				VALUE: "value3"
			}
		];

		Wade.setRatio();
		Wade.JWC.process();
	</script>
</body>
</html>
```

Touch UI中 JWC 控件是在组件属性中使用 ``ognl:`` 绑定数据集，“ds”是可以自定义的，能与定义数据集名称一致就行。

## 四、注意事项
+ JWC 组件中 “name”，不要重复；
+ Touch UI 框架的前端资源，包括样式和 JS 文件一定要路径准确；
+ 你的 HTML 里不能少了以下代码，且使用 ognl 绑定的数据定义必须是在组件解析之前，要在 Wade.JWC.process(); 执行前定义好。
```js
<script type="text/javascript">
// 这里写数据
Wade.setRatio();
Wade.JWC.process();
</script>
```
